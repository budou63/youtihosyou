Option Explicit

'==================== 基本設定 ====================
Private Const HEADER_ROW As Long = 3
Private Const DATA_START_ROW As Long = 4

' 参照元
Private Const SRC_FILE As String = "工事及び補助金改.xlsm"
Private Const SRC_SHEET As String = "表"

' 転記先
Private Const DST_SHEET As String = "補償明細"
Private Const TRIG_ADDRS As String = "P1"

' 出力開始行（縦詰め）
Private Const OUT_START_ROW As Long = 11
Private Const OUT_BASE_END_ROW As Long = 16
'==================================================


'==================== メイン ====================
Public Sub 補償明細_移転補償種類_内訳額_住所業者地番_転記()

    Dim wbSrc As Workbook, wsSrc As Worksheet
    Dim wsDst As Worksheet
    Dim key As String
    Dim colKey As Long
    Dim foundRow As Long, lastSrcRow As Long
    Dim foundSingleRow As Long

    Dim i As Long, colType As Long, colAmt As Long
    Dim singleKey As String
    Dim subjectText As String
    Dim addressText As String
    Dim vendorText As String
    Dim chibanText As String

    Dim hitAny As Boolean
    Dim msg As String

    Dim prevSU As Boolean, prevEE As Boolean, prevCalc As XlCalculation
    prevSU = Application.ScreenUpdating
    prevEE = Application.EnableEvents
    prevCalc = Application.Calculation

    On Error GoTo CleanFail
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    Set wsDst = ThisWorkbook.Worksheets(DST_SHEET)
    Set wbSrc = GetOrOpenWorkbook(SRC_FILE)
    Set wsSrc = wbSrc.Worksheets(SRC_SHEET)

    colKey = FindHeaderCol(wsSrc, "転記", HEADER_ROW)
    If colKey = 0 Then Err.Raise vbObjectError + 1, , "参照元「表」で見出し「転記」が見つかりません"

    lastSrcRow = wsSrc.Cells(wsSrc.rows.count, colKey).End(xlUp).Row
    If lastSrcRow < DATA_START_ROW Then GoTo CleanExit

    '--- 初期化（毎回クリア） ---
    ClearSingleFields wsDst
    ClearVerticalArea wsDst

    '========================
    ' 単発転記（P1のみ）
    '========================
    singleKey = Trim$(CStr(wsDst.Range("P1").Value))
    If singleKey = "" Then
        GoTo CleanExit
    End If

    foundSingleRow = FindRowByKey(wsSrc, colKey, singleKey, DATA_START_ROW, lastSrcRow)
    If foundSingleRow = 0 Then
        msg = msg & "・キー[" & singleKey & "] が参照元「転記」列に見つからず (P1)" & vbCrLf
        GoTo CleanExit
    End If

    ' --- 件名の転記（A2は加工なし、A5は加工あり） ---
    Dim rawSubject As String
    rawSubject = GetValueByHeader(wsSrc, foundSingleRow, "件名")
    
    ' A2: 未加工の件名
    If rawSubject <> "" Then
        WriteMergedTopLeft wsDst.Range("A2"), rawSubject
        hitAny = True
    End If
    
    ' A5: 加工後の件名（定型文言削除）
    subjectText = CleanTransferText(rawSubject)
    If subjectText <> "" Then
        WriteMergedTopLeft wsDst.Range("A5"), subjectText
        hitAny = True
    End If

    addressText = CleanTransferText(GetValueByHeader(wsSrc, foundSingleRow, "住所"))
    If addressText <> "" Then
        WriteMergedTopLeft wsDst.Range("G5"), addressText
        hitAny = True
    End If

    vendorText = CleanTransferText(GetValueByHeader(wsSrc, foundSingleRow, "施工業者・債権者"))
    If vendorText <> "" Then
        WriteMergedTopLeft wsDst.Range("L5"), vendorText
        hitAny = True
    End If

    chibanText = BuildAddressForC5(wsSrc, foundSingleRow)
    If chibanText <> "" Then
        WriteMergedTopLeft wsDst.Range("C5"), chibanText
        hitAny = True
    End If

    '========================
    ' 縦詰め転記（P1のキーで一致した行を対象）
    '========================
    key = Trim$(CStr(wsDst.Range("P1").Value))
    If key = "" Then
        GoTo CleanExit
    End If

    foundRow = FindRowByKey(wsSrc, colKey, key, DATA_START_ROW, lastSrcRow)
    If foundRow = 0 Then
        msg = msg & "・キー[" & key & "] が参照元「転記」列に見つからず" & vbCrLf
        GoTo CleanExit
    End If

    '========================
    ' 1) 縦詰め：移転補償種類 / 内訳額 / 数量（①～⑥）
    '========================
    Dim currentRowBase As Long
    Dim bifStartRow As Long
    Dim bifEndRow As Long
    Dim bifStarted As Boolean
    currentRowBase = OUT_START_ROW
    bifStartRow = OUT_START_ROW
    bifEndRow = OUT_START_ROW - 1
    bifStarted = False

    For i = 1 To 6
        Dim vType As String, vAmt As String, vQty As String
        vType = ""
        vAmt = ""
        vQty = ""

        '--- 移転補償種類（B列） ---
        colType = FindHeaderCol(wsSrc, "移転補償種類" & CircleNum(i), HEADER_ROW)
        If colType = 0 Then colType = FindHeaderCol(wsSrc, "移転補償種類" & CStr(i), HEADER_ROW)
        If colType > 0 Then
            vType = Trim$(CStr(wsSrc.Cells(foundRow, colType).Value))
        End If

        '--- 内訳額（I列） ---
        colAmt = FindHeaderCol(wsSrc, "内訳額" & CircleNum(i), HEADER_ROW)
        If colAmt = 0 Then colAmt = FindHeaderCol(wsSrc, "内訳額" & CStr(i), HEADER_ROW)
        If colAmt > 0 Then
            vAmt = Trim$(CStr(wsSrc.Cells(foundRow, colAmt).Value))
        End If

        '--- 数量（F列） ---
        Dim colQty As Long
        colQty = FindHeaderCol(wsSrc, "数量" & CircleNum(i), HEADER_ROW)
        If colQty = 0 Then colQty = FindHeaderCol(wsSrc, "数量" & CStr(i), HEADER_ROW)
        If colQty > 0 Then
            vQty = Trim$(CStr(wsSrc.Cells(foundRow, colQty).Value))
        End If

        If vType <> "" Or vAmt <> "" Or vQty <> "" Then
            InsertRowIfAnyValue wsDst, currentRowBase, OUT_BASE_END_ROW, Array("B", "I", "F")

            If Not bifStarted Then
                bifStartRow = currentRowBase
                bifStarted = True
            End If

            If vType <> "" Then
                WriteMergedTopLeft wsDst.Cells(currentRowBase, "B"), vType
                hitAny = True
            End If

            If vAmt <> "" Then
                WriteMergedTopLeft wsDst.Cells(currentRowBase, "I"), vAmt
                hitAny = True
            End If

            If vQty <> "" Then
                WriteMergedTopLeft wsDst.Cells(currentRowBase, "F"), vQty
                hitAny = True
            End If

            Dim maxHeight As Long
            maxHeight = GetMaxHeightForRow(wsDst, currentRowBase, Array("B", "I", "F"))
            bifEndRow = currentRowBase + maxHeight - 1
            currentRowBase = currentRowBase + maxHeight
        End If

    Next i

    '========================
    ' 2) 契約年月日 / 申出日 / 竣工年月日等（P～W列）
    '========================
    Dim colKeiyaku As Long, vKeiyaku As String
    colKeiyaku = FindHeaderCol(wsSrc, "契約年月日", HEADER_ROW)
    If colKeiyaku > 0 Then vKeiyaku = Trim$(CStr(wsSrc.Cells(foundRow, colKeiyaku).Value))

    Dim colMoushutsu As Long, vMoushutsu As String
    colMoushutsu = FindHeaderCol(wsSrc, "申出日", HEADER_ROW)
    If colMoushutsu > 0 Then vMoushutsu = Trim$(CStr(wsSrc.Cells(foundRow, colMoushutsu).Value))

    Dim colShunko As Long, vShunko As String
    colShunko = FindHeaderCol(wsSrc, "竣工年月日", HEADER_ROW)
    If colShunko > 0 Then vShunko = Trim$(CStr(wsSrc.Cells(foundRow, colShunko).Value))

    Dim colShunko2 As Long, vShunko2 As String
    colShunko2 = FindHeaderCol(wsSrc, "竣工年月日", HEADER_ROW)
    If colShunko2 > 0 Then vShunko2 = Trim$(CStr(wsSrc.Cells(foundRow, colShunko2).Value))

    Dim colKensa As Long, vKensa As String
    colKensa = FindHeaderCol(wsSrc, "検査・登記確認年月日", HEADER_ROW)
    If colKensa > 0 Then vKensa = Trim$(CStr(wsSrc.Cells(foundRow, colKensa).Value))

    Dim colKensain As Long, vKensain As String
    colKensain = FindHeaderCol(wsSrc, "検査員", HEADER_ROW)
    If colKensain > 0 Then vKensain = Trim$(CStr(wsSrc.Cells(foundRow, colKensain).Value))

    Dim colStartFinal As Long, vStartFinal As String
    colStartFinal = FindHeaderColByCandidates(wsSrc, Array("始（最終）", "始" & vbLf & "（最終）"), HEADER_ROW)
    If colStartFinal > 0 Then vStartFinal = Trim$(CStr(wsSrc.Cells(foundRow, colStartFinal).Value))

    Dim colEndFinal As Long, vEndFinal As String
    colEndFinal = FindHeaderColByCandidates(wsSrc, Array("終（最終）", "終" & vbLf & "（最終）"), HEADER_ROW)
    If colEndFinal > 0 Then vEndFinal = Trim$(CStr(wsSrc.Cells(foundRow, colEndFinal).Value))

    If bifEndRow >= bifStartRow Then
        Dim r As Long
        For r = bifStartRow To bifEndRow
            If vKeiyaku <> "" Then WriteMergedTopLeft wsDst.Cells(r, "P"), vKeiyaku
            If vMoushutsu <> "" Then WriteMergedTopLeft wsDst.Cells(r, "Q"), vMoushutsu
            If vShunko <> "" Then WriteMergedTopLeft wsDst.Cells(r, "R"), vShunko
            If vShunko2 <> "" Then WriteMergedTopLeft wsDst.Cells(r, "S"), vShunko2
            If vKensa <> "" Then WriteMergedTopLeft wsDst.Cells(r, "T"), vKensa
            If vKensain <> "" Then WriteMergedTopLeft wsDst.Cells(r, "U"), vKensain
            If vStartFinal <> "" Then WriteMergedTopLeft wsDst.Cells(r, "V"), vStartFinal
            If vEndFinal <> "" Then WriteMergedTopLeft wsDst.Cells(r, "W"), vEndFinal
        Next r
    End If

    '========================
    ' 3) 縦詰め転記：5,000万控除対象（①～⑥）
    '========================
    Dim currentRowO As Long
    currentRowO = OUT_START_ROW

    For i = 1 To 6
        Dim headerName As String
        headerName = "5,000万控除対象" & CircleNum(i)
        Dim colKoujo As Long
        colKoujo = FindHeaderCol(wsSrc, headerName, HEADER_ROW)
        Dim vKoujo As String
        vKoujo = ""
        If colKoujo > 0 Then vKoujo = Trim$(CStr(wsSrc.Cells(foundRow, colKoujo).Value))
        
        If vKoujo <> "" Then
            InsertRowIfAnyValue wsDst, currentRowO, OUT_BASE_END_ROW, Array("O")
            WriteMergedTopLeft wsDst.Cells(currentRowO, "O"), vKoujo
            hitAny = True
            Dim maxHeightO As Long
            maxHeightO = GetMaxHeightForRow(wsDst, currentRowO, Array("O"))
            currentRowO = currentRowO + maxHeightO
        End If
    Next i

    If Not hitAny Then
        MsgBox "転記対象が見つからなかったため、値は出力されませんでした。" & vbCrLf & vbCrLf & msg, vbExclamation
    End If

CleanExit:
    Application.ScreenUpdating = prevSU
    Application.EnableEvents = prevEE
    Application.Calculation = prevCalc
    Exit Sub

CleanFail:
    Application.ScreenUpdating = prevSU
    Application.EnableEvents = prevEE
    Application.Calculation = prevCalc
    MsgBox "転記中にエラー: " & Err.Description, vbCritical
End Sub

'==================== 以降、サブ関数（変更なし） ====================

Public Sub 補償明細_クリア()
    Dim wsDst As Worksheet
    Dim prevSU As Boolean, prevEE As Boolean, prevCalc As XlCalculation
    prevSU = Application.ScreenUpdating
    prevEE = Application.EnableEvents
    prevCalc = Application.Calculation
    On Error GoTo CleanExit
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual
    Set wsDst = ThisWorkbook.Worksheets(DST_SHEET)
    If Not wsDst.Range("P1").HasFormula Then wsDst.Range("P1").ClearContents
    ClearSingleFields wsDst
    ClearVerticalArea wsDst
CleanExit:
    Application.ScreenUpdating = prevSU
    Application.EnableEvents = prevEE
    Application.Calculation = prevCalc
    ThisWorkbook.Worksheets(DST_SHEET).Range("P1").Select
End Sub

Private Function BuildAddressForC5(ByVal wsSrc As Worksheet, ByVal srcRow As Long) As String
    Dim oaza As String, chiban1 As String, chiban2 As String, chiban3 As String
    Dim s As String, hasMore As Boolean
    oaza = Trim$(GetValueByHeader(wsSrc, srcRow, "大字①"))
    chiban1 = NormalizeChiban(Trim$(GetValueByHeader(wsSrc, srcRow, "元地番①")))
    chiban2 = NormalizeChiban(Trim$(GetValueByHeader(wsSrc, srcRow, "元地番②")))
    chiban3 = Trim$(GetValueByHeader(wsSrc, srcRow, "元地番③"))
    If chiban3 <> "" Then hasMore = True
    If oaza = "" And chiban1 = "" And chiban2 = "" Then
        BuildAddressForC5 = ""
        Exit Function
    End If
    s = "総社市"
    If oaza <> "" Then s = s & oaza
    If chiban1 <> "" Then s = s & chiban1
    If chiban2 <> "" Then
        If chiban1 <> "" Then s = s & "、"
        s = s & chiban2
    End If
    If hasMore Then s = s & "外"
    BuildAddressForC5 = s
End Function

Private Function NormalizeChiban(ByVal s As String) As String
    If s = "" Then NormalizeChiban = "": Exit Function
    If InStr(s, "-") > 0 Or InStr(s, "－") > 0 Or InStr(s, "―") > 0 Or InStr(s, "‐") > 0 Then
        s = Replace(s, "－", "-"): s = Replace(s, "―", "-"): s = Replace(s, "‐", "-")
        s = Replace(s, "-", "番地")
        NormalizeChiban = s
    Else
        NormalizeChiban = s & "番地"
    End If
End Function

Private Function GetValueByHeader(ByVal ws As Worksheet, ByVal srcRow As Long, ByVal headerName As String) As String
    Dim c As Long
    c = FindHeaderCol(ws, headerName, HEADER_ROW)
    If c = 0 Then GetValueByHeader = "" Else GetValueByHeader = Trim$(CStr(ws.Cells(srcRow, c).Value))
End Function

Private Function GetOrOpenWorkbook(ByVal fileName As String) As Workbook
    Dim wb As Workbook, basePath As String, fullPath As String, altName As String, foundPath As String
    basePath = ThisWorkbook.path: fullPath = basePath & "\" & fileName
    For Each wb In Application.Workbooks
        If StrComp(wb.name, fileName, vbTextCompare) = 0 Then Set GetOrOpenWorkbook = wb: Exit Function
    Next wb
    if Dir(fullPath) = "" Then
        altName = Dir(basePath & "\*工事*補助金改*.xlsm")
        If altName <> "" Then foundPath = basePath & "\" & altName Else Err.Raise vbObjectError + 100, , "参照元ブックが見つかりません"
    Else
        foundPath = fullPath
    End If
    Set GetOrOpenWorkbook = Application.Workbooks.Open(foundPath, ReadOnly:=True)
End Function

Private Function FindHeaderCol(ByVal ws As Worksheet, ByVal headerName As String, ByVal headerRow As Long) As Long
    Dim lastCol As Long, c As Long
    lastCol = ws.Cells(headerRow, ws.Columns.count).End(xlToLeft).Column
    For c = 1 To lastCol
        If NormalizeHeaderText(CStr(ws.Cells(headerRow, c).Value)) = NormalizeHeaderText(headerName) Then FindHeaderCol = c: Exit Function
    Next c
    FindHeaderCol = 0
End Function

Private Function FindHeaderColByCandidates(ByVal ws As Worksheet, ByVal headerNames As Variant, ByVal headerRow As Long) As Long
    Dim i As Long, colNum As Long
    For i = LBound(headerNames) To UBound(headerNames)
        colNum = FindHeaderCol(ws, CStr(headerNames(i)), headerRow)
        If colNum > 0 Then FindHeaderColByCandidates = colNum: Exit Function
    Next i
    FindHeaderColByCandidates = 0
End Function

Private Function NormalizeHeaderText(ByVal s As String) As String
    s = Replace$(s, vbCr, ""): s = Replace$(s, vbLf, ""): s = Replace$(s, "　", " "): s = Trim$(s)
    Do While InStr(s, "  ") > 0: s = Replace$(s, "  ", " "): Loop
    s = Replace$(s, " ", "")
    NormalizeHeaderText = s
End Function

Private Function FindRowByKey(ByVal ws As Worksheet, ByVal keyCol As Long, ByVal key As String, ByVal sRow As Long, ByVal eRow As Long) As Long
    Dim f As Range
    Set f = ws.Range(ws.Cells(sRow, keyCol), ws.Cells(eRow, keyCol)).Find(What:=key, LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=False)
    If Not f Is Nothing Then FindRowByKey = f.Row Else FindRowByKey = 0
End Function

Private Sub ClearMergedRange(ByVal ws As Worksheet, ByVal colLetter As String, ByVal sRow As Long, ByVal eRow As Long)
    Dim r As Long, c As Range, ma As Range, colNum As Long
    colNum = ws.Columns(colLetter).Column: r = sRow
    Do While r <= eRow
        Set c = ws.Cells(r, colNum)
        If c.MergeCells Then
            Set ma = c.MergeArea
            If Not MergeAreaHasFormula(ma) Then ma.ClearContents
            r = ma.Row + ma.rows.count
        Else
            If Not c.HasFormula Then c.ClearContents
            r = r + 1
        End If
    Loop
End Sub

Private Sub ClearCellMerged(ByVal c As Range)
    If c.MergeCells Then
        If Not MergeAreaHasFormula(c.MergeArea) Then c.MergeArea.ClearContents
    Else
        If Not c.HasFormula Then c.ClearContents
    End If
End Sub

Private Sub ClearSingleFields(ByVal ws As Worksheet)
    ClearCellMerged ws.Range("A2"): ClearCellMerged ws.Range("A5")
    ClearCellMerged ws.Range("C5"): ClearCellMerged ws.Range("G5"): ClearCellMerged ws.Range("L5")
End Sub

Private Sub ClearVerticalArea(ByVal ws As Worksheet)
    Dim clearEnd As Long
    clearEnd = WorksheetFunction.Max(OUT_BASE_END_ROW, ws.Cells(ws.rows.count, "B").End(xlUp).Row, ws.Cells(ws.rows.count, "I").End(xlUp).Row, ws.Cells(ws.rows.count, "O").End(xlUp).Row, ws.Cells(ws.rows.count, "P").End(xlUp).Row)
    ClearMergedRange ws, "B", OUT_START_ROW, clearEnd: ClearMergedRange ws, "I", OUT_START_ROW, clearEnd
    ClearMergedRange ws, "F", OUT_START_ROW, clearEnd: ClearMergedRange ws, "P", OUT_START_ROW, clearEnd
    ClearMergedRange ws, "Q", OUT_START_ROW, clearEnd: ClearMergedRange ws, "R", OUT_START_ROW, clearEnd
    ClearMergedRange ws, "S", OUT_START_ROW, clearEnd: ClearMergedRange ws, "T", OUT_START_ROW, clearEnd
    ClearMergedRange ws, "U", OUT_START_ROW, clearEnd: ClearMergedRange ws, "V", OUT_START_ROW, clearEnd
    ClearMergedRange ws, "W", OUT_START_ROW, clearEnd: ClearMergedRange ws, "O", OUT_START_ROW, clearEnd
End Sub

Private Sub WriteMergedTopLeft(ByVal c As Range, ByVal v As String)
    If c.MergeCells Then c.MergeArea.Cells(1, 1).Value = v Else c.Value = v
End Sub

Private Function MergeAreaHasFormula(ByVal ma As Range) As Boolean
    Dim cell As Range
    For Each cell In ma.Cells: If cell.HasFormula Then MergeAreaHasFormula = True: Exit Function
    Next cell
End Function

Private Function GetMaxHeightForRow(ByVal ws As Worksheet, ByVal rowNum As Long, ByVal colLetters As Variant) As Long
    Dim i As Long, h As Long, maxH As Long, c As Range
    maxH = 1
    For i = LBound(colLetters) To UBound(colLetters)
        Set c = ws.Cells(rowNum, ws.Columns(CStr(colLetters(i))).Column)
        If c.MergeCells Then h = c.MergeArea.rows.count Else h = 1
        If h > maxH Then maxH = h
    Next i
    GetMaxHeightForRow = maxH
End Function

Private Function HasAnyValueInRow(ByVal ws As Worksheet, ByVal rowNum As Long, ByVal colLetters As Variant) As Boolean
    Dim i As Long, c As Range, v As String
    For i = LBound(colLetters) To UBound(colLetters)
        Set c = ws.Cells(rowNum, ws.Columns(CStr(colLetters(i))).Column)
        If c.MergeCells Then v = Trim$(CStr(c.MergeArea.Cells(1, 1).Value)) Else v = Trim$(CStr(c.Value))
        If v <> "" Then HasAnyValueInRow = True: Exit Function
    Next i
End Function

Private Sub InsertRowIfAnyValue(ByVal ws As Worksheet, ByVal rowNum As Long, ByVal baseEnd As Long, ByVal colLetters As Variant)
    If rowNum > baseEnd Then ws.rows(rowNum).Insert Shift:=xlShiftDown, CopyOrigin:=xlFormatFromLeftOrAbove: Exit Sub
    If HasAnyValueInRow(ws, rowNum, colLetters) Then ws.rows(rowNum).Insert Shift:=xlShiftDown, CopyOrigin:=xlFormatFromLeftOrAbove
End Sub

Private Function CircleNum(ByVal n As Long) As String
    Select Case n
        Case 1: CircleNum = "①": Case 2: CircleNum = "②": Case 3: CircleNum = "③"
        Case 4: CircleNum = "④": Case 5: CircleNum = "⑤": Case 6: CircleNum = "⑥"
        Case Else: CircleNum = CStr(n)
    End Select
End Function

Private Function CleanTransferText(ByVal s As String) As String
    If s = "" Then CleanTransferText = "": Exit Function
    s = Replace$(s, "に伴う物件移転補償費", ""): s = Replace$(s, "に伴う用地費", "")
    CleanTransferText = s
End Function
