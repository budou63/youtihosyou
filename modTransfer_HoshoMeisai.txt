Option Explicit

'==================== 基本設定 ====================
Private Const HEADER_ROW As Long = 3
Private Const DATA_START_ROW As Long = 4

' 参照元
Private Const SRC_FILE As String = "工事及び補助金改.xlsm"
Private Const SRC_SHEET As String = "表"

' 転記先
Private Const DST_SHEET As String = "補償明細"
Private Const TRIG_ADDRS As String = "P1"

' 出力開始行（縦詰め）
Private Const OUT_START_ROW As Long = 11
Private Const OUT_BASE_END_ROW As Long = 16
'==================================================


'==================== メイン ====================
Public Sub 補償明細_移転補償種類_内訳額_住所業者地番_転記()

    Dim wbSrc As Workbook, wsSrc As Worksheet
    Dim wsDst As Worksheet
    Dim key As String
    Dim colKey As Long
    Dim foundRow As Long, lastSrcRow As Long
    Dim foundSingleRow As Long

    Dim i As Long, colType As Long, colAmt As Long
    Dim singleKey As String
    Dim subjectText As String
    Dim addressText As String
    Dim vendorText As String
    Dim chibanText As String

    Dim hitAny As Boolean
    Dim msg As String

    Dim prevSU As Boolean, prevEE As Boolean, prevCalc As XlCalculation
    prevSU = Application.ScreenUpdating
    prevEE = Application.EnableEvents
    prevCalc = Application.Calculation

    On Error GoTo CleanFail
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    Set wsDst = ThisWorkbook.Worksheets(DST_SHEET)
    Set wbSrc = GetOrOpenWorkbook(SRC_FILE)
    Set wsSrc = wbSrc.Worksheets(SRC_SHEET)

    colKey = FindHeaderCol(wsSrc, "転記", HEADER_ROW)
    If colKey = 0 Then Err.Raise vbObjectError + 1, , "参照元「表」で見出し「転記」が見つかりません"

    lastSrcRow = wsSrc.Cells(wsSrc.Rows.Count, colKey).End(xlUp).Row
    If lastSrcRow < DATA_START_ROW Then GoTo CleanExit

    '--- 初期化（毎回クリア） ---
    ClearSingleFields wsDst
    ClearVerticalArea wsDst

    '========================
    ' 単発転記（P1のみ）
    '========================
    singleKey = Trim$(CStr(wsDst.Range("P1").Value))
    If singleKey = "" Then
        GoTo CleanExit
    End If

    foundSingleRow = FindRowByKey(wsSrc, colKey, singleKey, DATA_START_ROW, lastSrcRow)
    If foundSingleRow = 0 Then
        msg = msg & "・キー[" & singleKey & "] が参照元「転記」列に見つからず (P1)" & vbCrLf
        GoTo CleanExit
    End If

    subjectText = CleanTransferText(GetValueByHeader(wsSrc, foundSingleRow, "件名"))
    If subjectText <> "" Then
        WriteMergedTopLeft wsDst.Range("A5"), subjectText
        hitAny = True
    End If

    addressText = CleanTransferText(GetValueByHeader(wsSrc, foundSingleRow, "住所"))
    If addressText <> "" Then
        WriteMergedTopLeft wsDst.Range("G5"), addressText
        hitAny = True
    End If

    vendorText = CleanTransferText(GetValueByHeader(wsSrc, foundSingleRow, "施工業者・債権者"))
    If vendorText <> "" Then
        WriteMergedTopLeft wsDst.Range("L5"), vendorText
        hitAny = True
    End If

    chibanText = BuildAddressForC5(wsSrc, foundSingleRow)
    If chibanText <> "" Then
        WriteMergedTopLeft wsDst.Range("C5"), chibanText
        hitAny = True
    End If

    '========================
    ' 縦詰め転記（P1のキーで一致した行を対象）
    '========================
    key = Trim$(CStr(wsDst.Range("P1").Value))
    If key = "" Then
        GoTo CleanExit
    End If

    foundRow = FindRowByKey(wsSrc, colKey, key, DATA_START_ROW, lastSrcRow)
    If foundRow = 0 Then
        msg = msg & "・キー[" & key & "] が参照元「転記」列に見つからず" & vbCrLf
        GoTo CleanExit
    End If

    '========================
    ' 1) 縦詰め：移転補償種類 / 内訳額 / 数量（①～⑥）
    '【修正】セット単位で行挿入判定を行い、1回だけ行挿入する
    '========================
    Dim currentRowBase As Long
    currentRowBase = OUT_START_ROW

    For i = 1 To 6

        ' このセット（①～⑥の1つ）で転記するデータを先に取得
        Dim vType As String, vAmt As String, vQty As String
        vType = ""
        vAmt = ""
        vQty = ""

        '--- 移転補償種類（B列） ---
        colType = FindHeaderCol(wsSrc, "移転補償種類" & CircleNum(i), HEADER_ROW)
        If colType = 0 Then colType = FindHeaderCol(wsSrc, "移転補償種類" & CStr(i), HEADER_ROW)
        If colType > 0 Then
            vType = Trim$(CStr(wsSrc.Cells(foundRow, colType).Value))
        End If

        '--- 内訳額（I列） ---
        colAmt = FindHeaderCol(wsSrc, "内訳額" & CircleNum(i), HEADER_ROW)
        If colAmt = 0 Then colAmt = FindHeaderCol(wsSrc, "内訳額" & CStr(i), HEADER_ROW)
        If colAmt > 0 Then
            vAmt = Trim$(CStr(wsSrc.Cells(foundRow, colAmt).Value))
        End If

        '--- 数量（F列） ---
        Dim colQty As Long
        colQty = FindHeaderCol(wsSrc, "数量" & CircleNum(i), HEADER_ROW)
        If colQty = 0 Then colQty = FindHeaderCol(wsSrc, "数量" & CStr(i), HEADER_ROW)
        If colQty > 0 Then
            vQty = Trim$(CStr(wsSrc.Cells(foundRow, colQty).Value))
        End If

        ' このセットにデータがある場合のみ転記
        If vType <> "" Or vAmt <> "" Or vQty <> "" Then
            '【修正】セット単位で行挿入判定（B/I/Fのいずれかに既存値があれば1回だけ行挿入）
            InsertRowIfAnyValue wsDst, currentRowBase, OUT_BASE_END_ROW, Array("B", "I", "F")

            ' 同じcurrentRowBaseにB/I/Fを書き込む
            If vType <> "" Then
                WriteMergedTopLeft wsDst.Cells(currentRowBase, "B"), vType
                hitAny = True
            End If

            If vAmt <> "" Then
                WriteMergedTopLeft wsDst.Cells(currentRowBase, "I"), vAmt
                hitAny = True
            End If

            If vQty <> "" Then
                WriteMergedTopLeft wsDst.Cells(currentRowBase, "F"), vQty
                hitAny = True
            End If

            ' この行のB/I/Fの最大高さを取得して、次のrowBaseに進む
            Dim maxHeight As Long
            maxHeight = GetMaxHeightForRow(wsDst, currentRowBase, Array("B", "I", "F"))
            currentRowBase = currentRowBase + maxHeight
        End If

    Next i

    '========================
    ' 2) 単一項目の縦詰め転記：契約年月日 / 申出日 / 竣工年月日
    '【修正】P/Q/Rを同一行に揃え、セット単位で1回だけ行挿入する
    '========================
    Dim outRowDate As Long
    outRowDate = OUT_START_ROW

    '--- 契約年月日（P列） ---
    Dim colKeiyaku As Long
    colKeiyaku = FindHeaderCol(wsSrc, "契約年月日", HEADER_ROW)
    Dim vKeiyaku As String
    vKeiyaku = ""
    If colKeiyaku > 0 Then
        vKeiyaku = Trim$(CStr(wsSrc.Cells(foundRow, colKeiyaku).Value))
    End If

    '--- 申出日（Q列） ---
    Dim colMoushutsu As Long
    colMoushutsu = FindHeaderCol(wsSrc, "申出日", HEADER_ROW)
    Dim vMoushutsu As String
    vMoushutsu = ""
    If colMoushutsu > 0 Then
        vMoushutsu = Trim$(CStr(wsSrc.Cells(foundRow, colMoushutsu).Value))
    End If

    '--- 竣工年月日（R列） ---
    Dim colShunko As Long
    colShunko = FindHeaderCol(wsSrc, "竣工年月日", HEADER_ROW)
    Dim vShunko As String
    vShunko = ""
    If colShunko > 0 Then
        vShunko = Trim$(CStr(wsSrc.Cells(foundRow, colShunko).Value))
    End If

    ' P/Q/Rのいずれかが存在する場合は、同一行に書き込む
    If vKeiyaku <> "" Or vMoushutsu <> "" Or vShunko <> "" Then
        '【修正】セット単位で行挿入判定（P/Q/Rのいずれかに既存値があれば1回だけ行挿入）
        InsertRowIfAnyValue wsDst, outRowDate, OUT_BASE_END_ROW, Array("P", "Q", "R")

        ' 同じoutRowDateにP/Q/Rを書き込む
        If vKeiyaku <> "" Then
            WriteMergedTopLeft wsDst.Cells(outRowDate, "P"), vKeiyaku
            hitAny = True
        End If

        If vMoushutsu <> "" Then
            WriteMergedTopLeft wsDst.Cells(outRowDate, "Q"), vMoushutsu
            hitAny = True
        End If

        If vShunko <> "" Then
            WriteMergedTopLeft wsDst.Cells(outRowDate, "R"), vShunko
            hitAny = True
        End If

        ' MaxHeight方式で次行に進む
        Dim dateMaxHeight As Long
        dateMaxHeight = GetMaxHeightForRow(wsDst, outRowDate, Array("P", "Q", "R"))
        outRowDate = outRowDate + dateMaxHeight
    End If

    If Not hitAny Then
        MsgBox "転記対象が見つからなかったため、値は出力されませんでした。" & vbCrLf & vbCrLf & msg, vbExclamation
    End If

CleanExit:
    Application.ScreenUpdating = prevSU
    Application.EnableEvents = prevEE
    Application.Calculation = prevCalc
    Exit Sub

CleanFail:
    Application.ScreenUpdating = prevSU
    Application.EnableEvents = prevEE
    Application.Calculation = prevCalc
    MsgBox "転記中にエラー: " & Err.Description, vbCritical

End Sub
'==================================================


'==================== クリアボタン用 ====================
Public Sub 補償明細_クリア()
    Dim wsDst As Worksheet
    Dim prevSU As Boolean, prevEE As Boolean, prevCalc As XlCalculation

    prevSU = Application.ScreenUpdating
    prevEE = Application.EnableEvents
    prevCalc = Application.Calculation

    On Error GoTo CleanExit
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    Set wsDst = ThisWorkbook.Worksheets(DST_SHEET)

    wsDst.Range("P1").ClearContents
    ClearSingleFields wsDst
    ClearVerticalArea wsDst

CleanExit:
    Application.ScreenUpdating = prevSU
    Application.EnableEvents = prevEE
    Application.Calculation = prevCalc
    ThisWorkbook.Worksheets(DST_SHEET).Range("P1").Select
End Sub

'==================== 単発：C5用文字列生成 ====================
Private Function BuildAddressForC5(ByVal wsSrc As Worksheet, ByVal srcRow As Long) As String
    Dim oaza As String
    Dim chiban1 As String, chiban2 As String
    Dim chiban3 As String
    Dim s As String
    Dim hasMore As Boolean

    oaza = Trim$(GetValueByHeader(wsSrc, srcRow, "大字①"))
    chiban1 = NormalizeChiban(Trim$(GetValueByHeader(wsSrc, srcRow, "元地番①")))
    chiban2 = NormalizeChiban(Trim$(GetValueByHeader(wsSrc, srcRow, "元地番②")))
    chiban3 = Trim$(GetValueByHeader(wsSrc, srcRow, "元地番③"))

    If chiban3 <> "" Then hasMore = True

    If oaza = "" And chiban1 = "" And chiban2 = "" Then
        BuildAddressForC5 = ""
        Exit Function
    End If

    s = "総社市"
    If oaza <> "" Then s = s & oaza

    If chiban1 <> "" Then
        s = s & chiban1
    End If

    If chiban2 <> "" Then
        If chiban1 <> "" Then s = s & "、"
        s = s & chiban2
    End If

    If hasMore Then s = s & "外"

    BuildAddressForC5 = s
End Function


' ハイフン→番地、無ければ末尾に番地
Private Function NormalizeChiban(ByVal s As String) As String
    If s = "" Then
        NormalizeChiban = ""
        Exit Function
    End If

    ' いろんなハイフンを統一的に扱う
    If InStr(s, "-") > 0 Or InStr(s, "－") > 0 Or InStr(s, "―") > 0 Or InStr(s, "‐") > 0 Then
        s = Replace(s, "－", "-")
        s = Replace(s, "―", "-")
        s = Replace(s, "‐", "-")
        s = Replace(s, "-", "番地")
        NormalizeChiban = s
    Else
        NormalizeChiban = s & "番地"
    End If
End Function

'==================== 共通：ヘッダ名から値取得 ====================
Private Function GetValueByHeader(ByVal ws As Worksheet, ByVal srcRow As Long, ByVal headerName As String) As String
    Dim c As Long
    c = FindHeaderCol(ws, headerName, HEADER_ROW)
    If c = 0 Then
        GetValueByHeader = ""
    Else
        GetValueByHeader = Trim$(CStr(ws.Cells(srcRow, c).Value))
    End If
End Function

'==================== 共通ヘルパー ====================

Private Function GetOrOpenWorkbook(ByVal fileName As String) As Workbook
    Dim wb As Workbook
    Dim basePath As String
    Dim fullPath As String
    Dim altName As String
    Dim foundPath As String

    basePath = ThisWorkbook.Path
    fullPath = basePath & "\" & fileName

    For Each wb In Application.Workbooks
        If StrComp(wb.Name, fileName, vbTextCompare) = 0 Then
            Set GetOrOpenWorkbook = wb
            Exit Function
        End If
    Next wb

    If Dir(fullPath) = "" Then
        altName = Dir(basePath & "\*工事*補助金改*.xlsm")
        If altName <> "" Then
            foundPath = basePath & "\" & altName
        Else
            Err.Raise vbObjectError + 100, , "参照元ブックが見つかりません: " & vbCrLf & fullPath
        End If
    Else
        foundPath = fullPath
    End If

    Set GetOrOpenWorkbook = Application.Workbooks.Open(foundPath, ReadOnly:=True)
End Function


Private Function FindHeaderCol(ByVal ws As Worksheet, ByVal headerName As String, ByVal headerRow As Long) As Long
    Dim lastCol As Long, c As Long
    lastCol = ws.Cells(headerRow, ws.Columns.Count).End(xlToLeft).Column

    For c = 1 To lastCol
        If Trim$(CStr(ws.Cells(headerRow, c).Value)) = headerName Then
            FindHeaderCol = c
            Exit Function
        End If
    Next c
    FindHeaderCol = 0
End Function


Private Function FindRowByKey(ByVal ws As Worksheet, ByVal keyCol As Long, ByVal key As String, _
                             ByVal sRow As Long, ByVal eRow As Long) As Long
    Dim f As Range
    Set f = ws.Range(ws.Cells(sRow, keyCol), ws.Cells(eRow, keyCol)) _
             .Find(What:=key, LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=False)
    If Not f Is Nothing Then
        FindRowByKey = f.Row
    Else
        FindRowByKey = 0
    End If
End Function


'【修正】結合セル対応の効率的なクリア処理（rを進める方式）
Private Sub ClearMergedRange(ByVal ws As Worksheet, ByVal colLetter As String, ByVal sRow As Long, ByVal eRow As Long)
    Dim r As Long
    Dim c As Range, ma As Range
    Dim colNum As Long
    colNum = ws.Columns(colLetter).Column

    r = sRow
    Do While r <= eRow
        Set c = ws.Cells(r, colNum)
        If c.MergeCells Then
            Set ma = c.MergeArea
            ma.ClearContents
            r = ma.Row + ma.Rows.Count
        Else
            c.ClearContents
            r = r + 1
        End If
    Loop
End Sub


Private Sub ClearCellMerged(ByVal c As Range)
    If c.MergeCells Then
        c.MergeArea.ClearContents
    Else
        c.ClearContents
    End If
End Sub


Private Sub ClearSingleFields(ByVal ws As Worksheet)
    ClearCellMerged ws.Range("A5")
    ClearCellMerged ws.Range("C5")
    ClearCellMerged ws.Range("G5")
    ClearCellMerged ws.Range("L5")
End Sub


Private Sub ClearVerticalArea(ByVal ws As Worksheet)
    Dim lastRowB As Long, lastRowI As Long, lastRowF As Long
    Dim lastRowP As Long, lastRowQ As Long, lastRowR As Long
    Dim clearEnd As Long

    lastRowB = ws.Cells(ws.Rows.Count, "B").End(xlUp).Row
    lastRowI = ws.Cells(ws.Rows.Count, "I").End(xlUp).Row
    lastRowF = ws.Cells(ws.Rows.Count, "F").End(xlUp).Row
    lastRowP = ws.Cells(ws.Rows.Count, "P").End(xlUp).Row
    lastRowQ = ws.Cells(ws.Rows.Count, "Q").End(xlUp).Row
    lastRowR = ws.Cells(ws.Rows.Count, "R").End(xlUp).Row

    clearEnd = WorksheetFunction.Max(OUT_BASE_END_ROW, lastRowB, lastRowI, lastRowF, lastRowP, lastRowQ, lastRowR)

    ClearMergedRange ws, "B", OUT_START_ROW, clearEnd
    ClearMergedRange ws, "I", OUT_START_ROW, clearEnd
    ClearMergedRange ws, "F", OUT_START_ROW, clearEnd
    ClearMergedRange ws, "P", OUT_START_ROW, clearEnd
    ClearMergedRange ws, "Q", OUT_START_ROW, clearEnd
    ClearMergedRange ws, "R", OUT_START_ROW, clearEnd
End Sub


Private Sub WriteMergedTopLeft(ByVal c As Range, ByVal v As String)
    If c.MergeCells Then
        c.MergeArea.Cells(1, 1).Value = v
    Else
        c.Value = v
    End If
End Sub


Private Function GetMergeHeight(ByVal c As Range) As Long
    If c.MergeCells Then
        GetMergeHeight = c.MergeArea.Rows.Count
    Else
        GetMergeHeight = 1
    End If
End Function


'【追加】複数列の結合セル高さの最大値を取得
Private Function GetMaxHeightForRow(ByVal ws As Worksheet, ByVal rowNum As Long, ByVal colLetters As Variant) As Long
    Dim i As Long, h As Long, maxH As Long
    Dim colNum As Long
    Dim c As Range

    maxH = 1
    For i = LBound(colLetters) To UBound(colLetters)
        colNum = ws.Columns(CStr(colLetters(i))).Column
        Set c = ws.Cells(rowNum, colNum)
        If c.MergeCells Then
            h = c.MergeArea.Rows.Count
        Else
            h = 1
        End If
        If h > maxH Then maxH = h
    Next i

    GetMaxHeightForRow = maxH
End Function


'【追加】指定行の指定列のいずれかに値があるかチェック
Private Function HasAnyValueInRow(ByVal ws As Worksheet, ByVal rowNum As Long, ByVal colLetters As Variant) As Boolean
    Dim i As Long
    Dim colNum As Long
    Dim c As Range
    Dim v As String

    For i = LBound(colLetters) To UBound(colLetters)
        colNum = ws.Columns(CStr(colLetters(i))).Column
        Set c = ws.Cells(rowNum, colNum)

        ' 結合セルの場合は左上セルを確認
        If c.MergeCells Then
            v = Trim$(CStr(c.MergeArea.Cells(1, 1).Value))
        Else
            v = Trim$(CStr(c.Value))
        End If

        If v <> "" Then
            HasAnyValueInRow = True
            Exit Function
        End If
    Next i

    HasAnyValueInRow = False
End Function


'【追加】指定行の指定列のいずれかに既存値があれば行挿入（セット単位で1回だけ）
Private Sub InsertRowIfAnyValue(ByVal ws As Worksheet, ByVal rowNum As Long, ByVal baseEnd As Long, ByVal colLetters As Variant)
    ' baseEndを超えている場合は行挿入
    If rowNum > baseEnd Then
        ws.Rows(rowNum).Insert Shift:=xlShiftDown, CopyOrigin:=xlFormatFromLeftOrAbove
        Exit Sub
    End If

    ' baseEnd以内でも、指定列のいずれかに既存値がある場合は行挿入
    If HasAnyValueInRow(ws, rowNum, colLetters) Then
        ws.Rows(rowNum).Insert Shift:=xlShiftDown, CopyOrigin:=xlFormatFromLeftOrAbove
    End If
End Sub


Private Function CircleNum(ByVal n As Long) As String
    Select Case n
        Case 1: CircleNum = "①"
        Case 2: CircleNum = "②"
        Case 3: CircleNum = "③"
        Case 4: CircleNum = "④"
        Case 5: CircleNum = "⑤"
        Case 6: CircleNum = "⑥"
        Case 7: CircleNum = "⑦"
        Case 8: CircleNum = "⑧"
        Case 9: CircleNum = "⑨"
        Case 10: CircleNum = "⑩"
        Case Else: CircleNum = CStr(n)
    End Select
End Function


Private Function CleanTransferText(ByVal s As String) As String
    If s = "" Then
        CleanTransferText = ""
        Exit Function
    End If

    s = Replace$(s, "に伴う物件移転補償費", "")
    s = Replace$(s, "に伴う用地費", "")
    CleanTransferText = s
End Function
