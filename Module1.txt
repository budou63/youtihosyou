Option Explicit

'==================== 設定（見出し・データ開始行） ====================
Private Const HEADER_ROW As Long = 3
Private Const DATA_START_ROW As Long = 4

'==================== 参照元（工事及び補助金改） ====================
Private Const SRC_SHEET As String = "表"
Private Const SRC_FILE_EXACT As String = "工事及び補助金改.xlsm"   ' あればこれを優先
Private Const SRC_FILE_PATTERN As String = "*工事*補助金改*.xlsm"  ' なければこれで探す

'==================== 参照先（土地調書） ====================
Private Const DST_SHEET As String = "用地明細"
Private Const TRIG_ADDRS As String = "S1,T1,U1"  ' トリガー入力セル（S→T→U）
Private Const REG_INFO_CELL As String = "J5"  ' B5=件名/F5=住所/M5=施工業者の並びに合わせ、同じ5行目の空き領域を登記事項用に確保
Private Const DEBUG_LOG As Boolean = False  ' ★FIX: X列 公募地目（必要時のみ True）

'==================== ブロック境界記録（B～Mの実出力結果に基づく） ====================
' 各トリガーの「実際に書き込んだ行」を Range として記録し、
' 後で S/T/U の転記範囲（blockStart～blockEnd）を確定するために使用する。
Private BlockRowsByTrigger As Collection

'==================== 5行目の値を一時保存 ====================
Private Saved_B5_Value As String
Private Saved_F5_Value As String
Private Saved_M5_Value As String
Private Saved_J5_Value As String

'======================================================================
' ③ 用地項目：筆①～⑩を同一行に揃えて出す（F/G/I/K/M ）
'     F:現況地目  G:公募地積  I:実測地積(実測地籍も吸収)  K:単価  M:買収金額
'======================================================================
Public Sub 用地項目縦詰め転記_TUV_FGIKM()

    Dim wbSrc As Workbook, wsSrc As Worksheet
    Dim wsDst As Worksheet
    Dim prevScreenUpdating As Boolean, prevEnableEvents As Boolean, prevCalc As XlCalculation
    Dim colTransfer As Long, lastSrcRow As Long, foundRow As Long
    Dim trigCells As Variant, addr As Variant, key As String
    Dim outRow As Long, baseEnd As Long
    Dim suffixes As Variant, i As Long
    Dim vGenkyo As String, vKobo As String, vJisoku As String, vTanka As String, vKaisyu As String
    Dim colGenkyo As Long, colKobo As Long, colJisoku As Long, colTanka As Long, colKaisyu As Long
    Dim h As Long
    Dim formulaM22 As String

    prevScreenUpdating = Application.ScreenUpdating
    prevEnableEvents = Application.EnableEvents
    prevCalc = Application.Calculation

    On Error GoTo CleanFail
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    Set wsDst = ThisWorkbook.Worksheets(DST_SHEET)
    formulaM22 = GetFormulaMergedSafe(wsDst.Range("M22"))

    ' 【重要な修正】F5の値を保存してからクリア処理
    Dim rngF5 As Range
    Set rngF5 = wsDst.Range("F5")
    Dim savedF5Value As String
    If rngF5.MergeCells Then
        savedF5Value = CStr(rngF5.MergeArea.Cells(1, 1).Value)
    Else
        savedF5Value = CStr(rngF5.Value)
    End If

    ' 【修正①】クリア：実際の最終行までクリア（残骸防止）
    baseEnd = 20
    Dim lastRowF As Long, lastRowG As Long, lastRowI As Long, lastRowK As Long, lastRowM As Long
    Dim clearEndF As Long, clearEndG As Long, clearEndI As Long, clearEndK As Long, clearEndM As Long
    lastRowF = wsDst.Cells(wsDst.rows.count, "F").End(xlUp).Row
    lastRowG = wsDst.Cells(wsDst.rows.count, "G").End(xlUp).Row
    lastRowI = wsDst.Cells(wsDst.rows.count, "I").End(xlUp).Row
    lastRowK = wsDst.Cells(wsDst.rows.count, "K").End(xlUp).Row
    lastRowM = wsDst.Cells(wsDst.rows.count, "M").End(xlUp).Row
    clearEndF = WorksheetFunction.Max(baseEnd, lastRowF)
    clearEndG = WorksheetFunction.Max(baseEnd, lastRowG)
    clearEndI = WorksheetFunction.Max(baseEnd, lastRowI)
    clearEndK = WorksheetFunction.Max(baseEnd, lastRowK)
    clearEndM = WorksheetFunction.Max(baseEnd, lastRowM)
    ClearOutputMerged wsDst, "F", 11, clearEndF
    ClearOutputMerged wsDst, "G", 11, clearEndG
    ClearOutputMerged wsDst, "I", 11, clearEndI
    ClearOutputMerged wsDst, "K", 11, clearEndK
    ClearOutputMerged wsDst, "M", 11, clearEndM
    RestoreFormulaMergedSafe wsDst.Range("M22"), formulaM22
    outRow = 11

    ' 【重要な修正】F5の値を復元
    If savedF5Value <> "" Then
        If rngF5.MergeCells Then
            rngF5.MergeArea.Cells(1, 1).Value = savedF5Value
        Else
            rngF5.Value = savedF5Value
        End If
    End If

    Set wbSrc = GetOrOpenSourceWorkbook(SRC_FILE_EXACT, SRC_FILE_PATTERN)
    Set wsSrc = wbSrc.Worksheets(SRC_SHEET)

    colTransfer = FindHeaderCol(wsSrc, "転記", HEADER_ROW)
    If colTransfer = 0 Then Err.Raise vbObjectError + 100, , "参照元「表」シートに見出し「転記」が見つかりません。"

    lastSrcRow = wsSrc.Cells(wsSrc.rows.count, colTransfer).End(xlUp).Row
    If lastSrcRow < DATA_START_ROW Then GoTo CleanExit

    suffixes = Array("①", "②", "③", "④", "⑤", "⑥", "⑦", "⑧", "⑨", "⑩")

    trigCells = Split(TRIG_ADDRS, ",")
    For Each addr In trigCells
        key = NormalizeKey(wsDst.Range(CStr(addr)).Value)
        If key <> "" Then
            foundRow = FindRowByKey(wsSrc, colTransfer, key, DATA_START_ROW, lastSrcRow)
            ' Debug.Print "key=", key, "foundRow=", foundRow ' ★FIX: X列 公募地目（必要なら有効化）
            If foundRow > 0 Then

                For i = LBound(suffixes) To UBound(suffixes)

                    colGenkyo = FindHeaderCol(wsSrc, "現況地目" & CStr(suffixes(i)), HEADER_ROW)
                    colKobo = FindHeaderCol(wsSrc, "公募地積" & CStr(suffixes(i)), HEADER_ROW)

                    colJisoku = FindHeaderCol(wsSrc, "実測地積" & CStr(suffixes(i)), HEADER_ROW)
                    If colJisoku = 0 Then colJisoku = FindHeaderCol(wsSrc, "実測地籍" & CStr(suffixes(i)), HEADER_ROW)

                    colTanka = FindHeaderCol(wsSrc, "単価" & CStr(suffixes(i)), HEADER_ROW)
                    colKaisyu = FindHeaderCol(wsSrc, "買収金額" & CStr(suffixes(i)), HEADER_ROW)

                    vGenkyo = IIf(colGenkyo > 0, CleanText(CStr(wsSrc.Cells(foundRow, colGenkyo).Value)), "")
                    vKobo = IIf(colKobo > 0, CleanText(CStr(wsSrc.Cells(foundRow, colKobo).Value)), "")
                    vJisoku = IIf(colJisoku > 0, CleanText(CStr(wsSrc.Cells(foundRow, colJisoku).Value)), "")
                    vTanka = IIf(colTanka > 0, CleanText(CStr(wsSrc.Cells(foundRow, colTanka).Value)), "")
                    vKaisyu = IIf(colKaisyu > 0, CleanText(CStr(wsSrc.Cells(foundRow, colKaisyu).Value)), "")

                    If (vGenkyo & vKobo & vJisoku & vTanka & vKaisyu) <> "" Then

                        EnsureWritableRowMulti wsDst, outRow, baseEnd, Array("F", "G", "I", "K", "M")

                        If vGenkyo <> "" Then WriteToMergedTopLeft wsDst, "F", outRow, vGenkyo
                        If vKobo <> "" Then WriteToMergedTopLeft wsDst, "G", outRow, vKobo
                        If vJisoku <> "" Then WriteToMergedTopLeft wsDst, "I", outRow, vJisoku
                        If vTanka <> "" Then WriteToMergedTopLeft wsDst, "K", outRow, vTanka
                        If vKaisyu <> "" Then WriteToMergedTopLeft wsDst, "M", outRow, vKaisyu
                        RecordBlockRow key, wsDst, outRow

                        h = GetMergedHeightMulti(wsDst, outRow, Array("F", "G", "I", "K", "M"))
                        outRow = outRow + h
                    End If

                Next i
            End If
        End If
    Next addr

CleanExit:
    ' 【重要な修正】M22の集計用SUM関数を復元（念のため）
    RestoreFormulaMergedSafe wsDst.Range("M22"), formulaM22

    ' 【重要な修正】最後にもう一度F5の値を復元（念のため）
    If savedF5Value <> "" Then
        Set rngF5 = wsDst.Range("F5")
        If rngF5.MergeCells Then
            rngF5.MergeArea.Cells(1, 1).Value = savedF5Value
        Else
            rngF5.Value = savedF5Value
        End If
    End If
    
    Application.ScreenUpdating = prevScreenUpdating
    Application.EnableEvents = prevEnableEvents
    Application.Calculation = prevCalc
    Exit Sub

CleanFail:
    If Not wsDst Is Nothing Then
        RestoreFormulaMergedSafe wsDst.Range("M22"), formulaM22
    End If
    Application.ScreenUpdating = prevScreenUpdating
    Application.EnableEvents = prevEnableEvents
    Application.Calculation = prevCalc
    MsgBox "転記中にエラー: " & Err.Description, vbExclamation
End Sub

'======================================================================
' 件名転記：B5 に参照元「表」シートの「件名」を転記
'======================================================================
Private Sub 件名転記_B5()

    Dim wbSrc As Workbook, wsSrc As Worksheet
    Dim wsDst As Worksheet
    Dim key As String
    Dim colTransfer As Long, colKenmei As Long
    Dim lastSrcRow As Long, foundRow As Long
    Dim kenmei As String
    Dim rngB5 As Range

    On Error GoTo CleanFail

    Set wsDst = ThisWorkbook.Worksheets(DST_SHEET)
    Set rngB5 = wsDst.Range("B5")

    key = GetFirstTriggerKey(wsDst)

    If key = "" Then
        If rngB5.MergeCells Then rngB5.MergeArea.ClearContents Else rngB5.ClearContents
        Saved_B5_Value = ""
        GoTo CleanExit
    End If

    Set wbSrc = GetOrOpenSourceWorkbook(SRC_FILE_EXACT, SRC_FILE_PATTERN)
    Set wsSrc = wbSrc.Worksheets(SRC_SHEET)

    colTransfer = FindHeaderCol(wsSrc, "転記", HEADER_ROW)
    If colTransfer = 0 Then Err.Raise vbObjectError + 100, , "参照元「表」シートに見出し「転記」が見つかりません。"

    colKenmei = FindHeaderCol(wsSrc, "件名", HEADER_ROW)
    If colKenmei = 0 Then Err.Raise vbObjectError + 110, , "参照元「表」シートに見出し「件名」が見つかりません。"

    lastSrcRow = wsSrc.Cells(wsSrc.rows.count, colTransfer).End(xlUp).Row
    If lastSrcRow < DATA_START_ROW Then
        If rngB5.MergeCells Then rngB5.MergeArea.ClearContents Else rngB5.ClearContents
        Saved_B5_Value = ""
        GoTo CleanExit
    End If

    foundRow = FindRowByKey(wsSrc, colTransfer, key, DATA_START_ROW, lastSrcRow)
    If foundRow = 0 Then
        If rngB5.MergeCells Then rngB5.MergeArea.ClearContents Else rngB5.ClearContents
        Saved_B5_Value = ""
        GoTo CleanExit
    End If

    kenmei = CleanText(CStr(wsSrc.Cells(foundRow, colKenmei).Value))

    If kenmei <> "" Then
        WriteToMergedTopLeft wsDst, "B", 5, kenmei
        Saved_B5_Value = kenmei
    Else
        If rngB5.MergeCells Then rngB5.MergeArea.ClearContents Else rngB5.ClearContents
        Saved_B5_Value = ""
    End If

CleanExit:
    Exit Sub

CleanFail:
    If Not rngB5 Is Nothing Then
        If rngB5.MergeCells Then rngB5.MergeArea.ClearContents Else rngB5.ClearContents
    End If
    Saved_B5_Value = ""
    Exit Sub
End Sub

'======================================================================
' 登記事項転記：REG_INFO_CELL に参照元「表」シートの「登記事項」を転記
'======================================================================
Private Sub 登記事項転記_RegInfo()

    Dim wbSrc As Workbook, wsSrc As Worksheet
    Dim wsDst As Worksheet
    Dim key As String
    Dim colTransfer As Long, colRegInfo As Long
    Dim lastSrcRow As Long, foundRow As Long
    Dim regInfo As String
    Dim rngRegInfo As Range
    Dim colLetter As String
    Dim rowNum As Long

    On Error GoTo CleanFail

    Set wsDst = ThisWorkbook.Worksheets(DST_SHEET)
    Set rngRegInfo = wsDst.Range(REG_INFO_CELL)
    colLetter = Split(rngRegInfo.Address(False, False), "$")(0)
    rowNum = rngRegInfo.Row

    key = GetFirstTriggerKey(wsDst)

    If key = "" Then
        ClearCellContentsMergedSafe rngRegInfo
        Saved_J5_Value = ""
        GoTo CleanExit
    End If

    Set wbSrc = GetOrOpenSourceWorkbook(SRC_FILE_EXACT, SRC_FILE_PATTERN)
    Set wsSrc = wbSrc.Worksheets(SRC_SHEET)

    colTransfer = FindHeaderCol(wsSrc, "転記", HEADER_ROW)
    If colTransfer = 0 Then Err.Raise vbObjectError + 100, , "参照元「表」シートに見出し「転記」が見つかりません。"

    colRegInfo = FindHeaderCol(wsSrc, "登記事項", HEADER_ROW)
    If colRegInfo = 0 Then Err.Raise vbObjectError + 140, , "参照元「表」シートに見出し「登記事項」が見つかりません。"

    lastSrcRow = wsSrc.Cells(wsSrc.rows.count, colTransfer).End(xlUp).Row
    If lastSrcRow < DATA_START_ROW Then
        ClearCellContentsMergedSafe rngRegInfo
        Saved_J5_Value = ""
        GoTo CleanExit
    End If

    foundRow = FindRowByKey(wsSrc, colTransfer, key, DATA_START_ROW, lastSrcRow)
    If foundRow = 0 Then
        ClearCellContentsMergedSafe rngRegInfo
        Saved_J5_Value = ""
        GoTo CleanExit
    End If

    regInfo = CleanText(CStr(wsSrc.Cells(foundRow, colRegInfo).Value))

    If regInfo <> "" Then
        WriteToMergedTopLeft wsDst, colLetter, rowNum, regInfo
        Saved_J5_Value = regInfo
    Else
        ClearCellContentsMergedSafe rngRegInfo
        Saved_J5_Value = ""
    End If

CleanExit:
    Exit Sub

CleanFail:
    If Not rngRegInfo Is Nothing Then
        ClearCellContentsMergedSafe rngRegInfo
    End If
    Saved_J5_Value = ""
    Exit Sub
End Sub

'==================== クリア用：最終データ行を取得（指定列群の最大） ====================
Private Function GetLastDataRow(ByVal ws As Worksheet, ByVal colLetters As Variant, ByVal startRow As Long, ByVal defaultEndRow As Long) As Long
    Dim i As Long, colNum As Long
    Dim lastRowCol As Long
    Dim maxRow As Long

    maxRow = defaultEndRow
    For i = LBound(colLetters) To UBound(colLetters)
        colNum = ws.Columns(CStr(colLetters(i))).Column
        lastRowCol = ws.Cells(ws.rows.count, colNum).End(xlUp).Row
        If lastRowCol >= startRow Then
            If lastRowCol > maxRow Then maxRow = lastRowCol
        End If
    Next i

    GetLastDataRow = maxRow
End Function

'======================================================================
' クリア：用地明細（トリガー・単発・縦詰め出力をまとめてクリア）
'======================================================================
Public Sub 用地明細_クリア()

    Dim wsDst As Worksheet
    Dim prevScreenUpdating As Boolean, prevEnableEvents As Boolean, prevCalc As XlCalculation
    Dim lastRow As Long
    Dim formulaM22 As String

    Set wsDst = ThisWorkbook.Worksheets(DST_SHEET)
    formulaM22 = GetFormulaMergedSafe(wsDst.Range("M22"))

    prevScreenUpdating = Application.ScreenUpdating
    prevEnableEvents = Application.EnableEvents
    prevCalc = Application.Calculation

    On Error GoTo CleanFail
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    '--- トリガーセルをクリア ---
    wsDst.Range("S1").ClearContents
    wsDst.Range("T1").ClearContents
    wsDst.Range("U1").ClearContents

    '--- 単発セル（結合考慮）をクリア ---
    ClearCellContentsMergedSafe wsDst.Range("B5")
    ClearCellContentsMergedSafe wsDst.Range("F5")
    ClearCellContentsMergedSafe wsDst.Range("M5")
    ClearCellContentsMergedSafe wsDst.Range(REG_INFO_CELL)

    '--- 保存値もクリア ---
    Saved_B5_Value = ""
    Saved_F5_Value = ""
    Saved_M5_Value = ""
    Saved_J5_Value = ""

    '--- 出力部の最終行を算出（行挿入に対応） ---
    ' M22のSUM関数は保存・復元で保護し、M列の明細残骸も従来どおり最終行までクリアする。
    lastRow = GetLastDataRow(wsDst, Array("B", "C", "D", "F", "G", "I", "K", "M", _
                                            "S", "T", "U", "V", "W", "X", _
                                            "Z", "AA", "AB", "AC", "AD", "AE", _
                                            "AF", "AG", "AH", "AI", "AJ", "AK"), 11, 20)

    '--- 縦詰め出力をクリア ---
    ClearOutputMerged wsDst, "B", 11, lastRow
    ClearOutputMerged wsDst, "C", 11, lastRow
    ClearOutputMerged wsDst, "D", 11, lastRow
    ClearOutputMerged wsDst, "F", 11, lastRow
    ClearOutputMerged wsDst, "G", 11, lastRow
    ClearOutputMerged wsDst, "I", 11, lastRow
    ClearOutputMerged wsDst, "K", 11, lastRow
    ClearOutputMerged wsDst, "M", 11, lastRow
    RestoreFormulaMergedSafe wsDst.Range("M22"), formulaM22
    ClearOutputMerged wsDst, "S", 11, lastRow
    ClearOutputMerged wsDst, "T", 11, lastRow
    ClearOutputMerged wsDst, "U", 11, lastRow
    ClearOutputMerged wsDst, "V", 11, lastRow
    ClearOutputMerged wsDst, "W", 11, lastRow
    ClearOutputMerged wsDst, "X", 11, lastRow
    ClearOutputMerged wsDst, "Z", 11, lastRow
    ClearOutputMerged wsDst, "AA", 11, lastRow
    ClearOutputMerged wsDst, "AB", 11, lastRow

    Transfer_STU_Vertical_Reconstruct

'======================================================================
' 【方式A】S/T/U列を統合して全体再構築方式で転記
' 【修正】S1/T1/U1 を順に処理し、B～M列の出力ブロックに合わせて同じ値を繰り返し転記
'======================================================================
Public Sub Transfer_STU_Vertical_Reconstruct()
    Dim wbSrc As Workbook, wsSrc As Worksheet
    Dim wsDst As Worksheet
    Dim prevScreenUpdating As Boolean, prevEnableEvents As Boolean, prevCalc As XlCalculation
    Dim colTransfer As Long, lastSrcRow As Long, foundRow As Long
    Dim trigCells As Variant, addr As Variant, key As String
    Dim outRow As Long, currentRow As Long, blockStart As Long, blockEnd As Long
    Dim colKeiyaku As Long, colMoushutsu As Long, colShunko As Long, colRegInfo As Long, colKifu As Long
    Dim vKeiyaku As String, vMoushutsu As String, vShunko As String, vRegInfo As String, vKifu As String
    Dim colKouboChimokuList() As Long
    Dim vKouboChimoku As String
    Dim kouboIndex As Long
    Dim kouboHeaderMax As Long
    Dim kouboHasValue As Boolean
    Dim ownerHeaderNames As Variant, ownerDestCols As Variant, clearCols As Variant
    Dim colOwnerList() As Long
    Dim ownerIndex As Long
    Dim vOwner As String
    Dim clearCol As Variant
    Dim baseName As Variant
    Dim h As Long
    Dim outStartRow As Long, outBaseEndRow As Long

    outStartRow = 11
    outBaseEndRow = 20

    prevScreenUpdating = Application.ScreenUpdating
    prevEnableEvents = Application.EnableEvents
    prevCalc = Application.Calculation

    On Error GoTo CleanFail
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    Set wsDst = ThisWorkbook.Worksheets(DST_SHEET)

    '--- ① 出力エリア（S～X列、Z～AK列の転記範囲）を毎回すべてクリア ---
    Dim clearEnd As Long
    clearCols = Array("S", "T", "U", "V", "W", "X", _
                      "Z", "AA", "AB", "AC", "AD", "AE", _
                      "AF", "AG", "AH", "AI", "AJ", "AK")
    clearEnd = GetLastDataRow(wsDst, clearCols, outStartRow, outBaseEndRow)
    For Each clearCol In clearCols
        ClearOutputMerged wsDst, CStr(clearCol), outStartRow, clearEnd
    Next clearCol

    '--- ② 参照元シートの準備 ---
    Set wbSrc = GetOrOpenSourceWorkbook(SRC_FILE_EXACT, SRC_FILE_PATTERN)
    Set wsSrc = wbSrc.Worksheets(SRC_SHEET)

    colTransfer = FindHeaderCol(wsSrc, "転記", HEADER_ROW)
    If colTransfer = 0 Then Err.Raise vbObjectError + 100, , "参照元「表」シートに見出し「転記」が見つかりません。"

    lastSrcRow = wsSrc.Cells(wsSrc.rows.count, colTransfer).End(xlUp).Row
    If lastSrcRow < DATA_START_ROW Then GoTo CleanExit

    '--- 見出し列を検索（接尾辞なし） ---
    colKeiyaku = FindHeaderCol(wsSrc, "契約年月日", HEADER_ROW)
    colMoushutsu = FindHeaderCol(wsSrc, "申出日", HEADER_ROW)
    colShunko = FindHeaderCol(wsSrc, "竣工年月日", HEADER_ROW)
    colRegInfo = FindHeaderCol(wsSrc, "登記事項", HEADER_ROW)
    colKifu = FindHeaderCol(wsSrc, "寄附等理由事業", HEADER_ROW) ' ★FIX: W列用見出しを追加

    '--- Z～AK列：土地所有者・相続人情報（ブロック先頭行のみ転記） ---
    ownerHeaderNames = Array("土地所有者住所", "土地所有者", _
                             "相続人住所①", "相続人①", _
                             "相続人住所②", "相続人②", _
                             "相続人住所③", "相続人③", _
                             "相続人住所④", "相続人④", _
                             "相続人住所⑤", "相続人⑤")
    ownerDestCols = Array("Z", "AA", "AB", "AC", "AD", "AE", "AF", "AG", "AH", "AI", "AJ", "AK")
    ReDim colOwnerList(LBound(ownerHeaderNames) To UBound(ownerHeaderNames))
    For ownerIndex = LBound(ownerHeaderNames) To UBound(ownerHeaderNames)
        colOwnerList(ownerIndex) = FindHeaderCol(wsSrc, CStr(ownerHeaderNames(ownerIndex)), HEADER_ROW)
    Next ownerIndex

    ' ★FIX: X列 公募地目（①～⑳の連番見出しに対応）
    kouboHeaderMax = 20
    ReDim colKouboChimokuList(1 To kouboHeaderMax)
    For kouboIndex = 1 To kouboHeaderMax
        colKouboChimokuList(kouboIndex) = 0
        For Each baseName In Array("公募地目", "公簿地目")
            colKouboChimokuList(kouboIndex) = FindHeaderCol(wsSrc, CStr(baseName) & CircleNumber(kouboIndex), HEADER_ROW)
            If colKouboChimokuList(kouboIndex) = 0 Then
                colKouboChimokuList(kouboIndex) = FindHeaderCol(wsSrc, CStr(baseName) & CStr(kouboIndex), HEADER_ROW)
            End If
            If colKouboChimokuList(kouboIndex) = 0 Then
                colKouboChimokuList(kouboIndex) = FindHeaderCol(wsSrc, CStr(baseName) & ToFullWidthNumber(kouboIndex), HEADER_ROW)
            End If
            If colKouboChimokuList(kouboIndex) > 0 Then Exit For
        Next baseName
        If DEBUG_LOG Then Debug.Print "kouboIndex=", kouboIndex, "col=", colKouboChimokuList(kouboIndex) ' ★FIX: X列 公募地目
    Next kouboIndex

    '--- ③ トリガーセルを順に処理し、B～Mの実出力ブロックに同期 ---
    trigCells = Split(TRIG_ADDRS, ",")

    For Each addr In trigCells
        key = NormalizeKey(wsDst.Range(CStr(addr)).Value)
        If key <> "" Then
            foundRow = FindRowByKey(wsSrc, colTransfer, key, DATA_START_ROW, lastSrcRow)
            If DEBUG_LOG Then Debug.Print "key=", key, "foundRow=", foundRow ' ★FIX: X列 公募地目
            If foundRow > 0 Then
                If GetBlockRangeByKey(wsDst, key, blockStart, blockEnd) Then
                    '--- ④ 該当データを取得（単一値） ---
                    vKeiyaku = ""
                    vMoushutsu = ""
                    vShunko = ""
                    vRegInfo = ""
                    vKifu = "" ' ★FIX: W列用

                    If colKeiyaku > 0 Then
                        vKeiyaku = CleanText(CStr(wsSrc.Cells(foundRow, colKeiyaku).Value))
                    End If
                    If colMoushutsu > 0 Then
                        vMoushutsu = CleanText(CStr(wsSrc.Cells(foundRow, colMoushutsu).Value))
                    End If
                    If colShunko > 0 Then
                        vShunko = CleanText(CStr(wsSrc.Cells(foundRow, colShunko).Value))
                    End If
                    If colRegInfo > 0 Then
                        vRegInfo = CleanTextKeepLineBreaks(CStr(wsSrc.Cells(foundRow, colRegInfo).Value))
                    End If
                    If colKifu > 0 Then
                        vKifu = CleanText(CStr(wsSrc.Cells(foundRow, colKifu).Value)) ' ★FIX: 寄附等理由事業を取得
                    End If
                    kouboHasValue = False
                    For kouboIndex = 1 To kouboHeaderMax
                        If colKouboChimokuList(kouboIndex) > 0 Then
                            vKouboChimoku = CleanText(CStr(wsSrc.Cells(foundRow, colKouboChimokuList(kouboIndex)).Value))
                            If vKouboChimoku <> "" Then
                                kouboHasValue = True
                                Exit For
                            End If
                        End If
                    Next kouboIndex

                    '--- Z～AK列はブロック先頭行（blockStart）にだけ1回転記 ---
                    For ownerIndex = LBound(ownerHeaderNames) To UBound(ownerHeaderNames)
                        vOwner = ""
                        If colOwnerList(ownerIndex) > 0 Then
                            vOwner = CleanText(CStr(wsSrc.Cells(foundRow, colOwnerList(ownerIndex)).Value))
                        End If
                        If vOwner <> "" Then
                            WriteToMergedTopLeft wsDst, CStr(ownerDestCols(ownerIndex)), blockStart, vOwner
                        End If
                    Next ownerIndex

                    '--- ⑤ 出力ブロック範囲にS～Xを同一値で繰り返し転記 ---
                    If (vKeiyaku <> "" Or vMoushutsu <> "" Or vShunko <> "" Or vRegInfo <> "" Or vKifu <> "" Or kouboHasValue) Then ' ★FIX: X列 公募地目を判定に追加
                        currentRow = blockStart
                        kouboIndex = 1
                        If DEBUG_LOG Then Debug.Print "blockStart=", blockStart, "blockEnd=", blockEnd ' ★FIX: X列 公募地目
                        Do While currentRow < blockEnd
                            EnsureWritableRowMulti wsDst, currentRow, outBaseEndRow, Array("S", "T", "U", "V", "W", "X") ' ★FIX: X列 公募地目を対象に追加

                            If vKeiyaku <> "" Then
                                WriteToMergedTopLeft wsDst, "S", currentRow, vKeiyaku
                            End If
                            If vMoushutsu <> "" Then
                                WriteToMergedTopLeft wsDst, "T", currentRow, vMoushutsu
                            End If
                            If vShunko <> "" Then
                                WriteToMergedTopLeft wsDst, "U", currentRow, vShunko
                            End If
                            If vRegInfo <> "" Then
                                WriteToMergedTopLeft wsDst, "V", currentRow, vRegInfo
                                SetWrapTextMerged wsDst, "V", currentRow, True
                            End If
                            If vKifu <> "" Then
                                WriteToMergedTopLeft wsDst, "W", currentRow, vKifu ' ★FIX: W列へ転記
                            End If
                            vKouboChimoku = ""
                            If kouboIndex <= kouboHeaderMax Then
                                If colKouboChimokuList(kouboIndex) > 0 Then
                                    vKouboChimoku = CleanText(CStr(wsSrc.Cells(foundRow, colKouboChimokuList(kouboIndex)).Value))
                                End If
                            End If
                            If vKouboChimoku <> "" Then
                                WriteToMergedTopLeft wsDst, "X", currentRow, vKouboChimoku ' ★FIX: X列 公募地目へ転記
                            End If
                            If DEBUG_LOG Then Debug.Print "kouboIndex=", kouboIndex, "col=", IIf(kouboIndex <= kouboHeaderMax, colKouboChimokuList(kouboIndex), 0), "value=", vKouboChimoku, "row=", currentRow ' ★FIX: X列 公募地目

                            h = GetMergedHeightMulti(wsDst, currentRow, Array("S", "T", "U", "V", "W", "X")) ' ★FIX: X列 公募地目を高さ判定に追加
                            currentRow = currentRow + h
                            kouboIndex = kouboIndex + 1
                        Loop
                    End If
                End If
            End If
        End If
    Next addr

CleanExit:
    Application.ScreenUpdating = prevScreenUpdating
    Application.EnableEvents = prevEnableEvents
    Application.Calculation = prevCalc
    Exit Sub

CleanFail:
    Application.ScreenUpdating = prevScreenUpdating
    Application.EnableEvents = prevEnableEvents
    Application.Calculation = prevCalc
    MsgBox "転記中にエラー: " & Err.Description, vbExclamation
End Sub

'======================================================================
' 共通：単一系列（例：大字①～⑩ / 字①～⑩ / 元地番①～⑩）を縦詰め転記
'======================================================================
Private Sub Transfer_OneHeaderSeries_Vertical( _
    ByVal outCol As String, ByVal outStartRow As Long, ByVal outBaseEndRow As Long, _
    ByVal headerPrefix As String, ByVal headerSuffixStyle As String, _
    ByVal replaceHyphenMode As String, _
    ByVal mergeRightCol As String)

    Dim wbSrc As Workbook, wsSrc As Worksheet
    Dim wsDst As Worksheet
    Dim prevScreenUpdating As Boolean, prevEnableEvents As Boolean, prevCalc As XlCalculation
    Dim colTransfer As Long, lastSrcRow As Long, foundRow As Long
    Dim trigCells As Variant, addr As Variant, key As String
    Dim outRow As Long
    Dim suffixes As Variant, i As Long
    Dim colTarget As Long
    Dim v As String

    prevScreenUpdating = Application.ScreenUpdating
    prevEnableEvents = Application.EnableEvents
    prevCalc = Application.Calculation

    On Error GoTo CleanFail
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    Set wsDst = ThisWorkbook.Worksheets(DST_SHEET)

    ' 【修正①】クリア：実際の最終行までクリア（残骸防止）
    Dim lastRowCol As Long, clearEnd As Long
    lastRowCol = wsDst.Cells(wsDst.rows.count, wsDst.Columns(outCol).Column).End(xlUp).Row
    clearEnd = WorksheetFunction.Max(outBaseEndRow, lastRowCol)
    ClearOutputMerged wsDst, outCol, outStartRow, clearEnd
    outRow = outStartRow

    Set wbSrc = GetOrOpenSourceWorkbook(SRC_FILE_EXACT, SRC_FILE_PATTERN)
    Set wsSrc = wbSrc.Worksheets(SRC_SHEET)

    colTransfer = FindHeaderCol(wsSrc, "転記", HEADER_ROW)
    If colTransfer = 0 Then Err.Raise vbObjectError + 100, , "参照元「表」シートに見出し「転記」が見つかりません。"

    lastSrcRow = wsSrc.Cells(wsSrc.rows.count, colTransfer).End(xlUp).Row
    If lastSrcRow < DATA_START_ROW Then GoTo CleanExit

    If headerSuffixStyle = "circled" Then
        suffixes = Array("①", "②", "③", "④", "⑤", "⑥", "⑦", "⑧", "⑨", "⑩")
    Else
        Err.Raise vbObjectError + 901, , "headerSuffixStyle が未対応です。"
    End If

    trigCells = Split(TRIG_ADDRS, ",")
    For Each addr In trigCells

        key = NormalizeKey(wsDst.Range(CStr(addr)).Value)
        If key <> "" Then

            foundRow = FindRowByKey(wsSrc, colTransfer, key, DATA_START_ROW, lastSrcRow)
            If foundRow > 0 Then

                For i = LBound(suffixes) To UBound(suffixes)

                    colTarget = FindHeaderCol(wsSrc, headerPrefix & CStr(suffixes(i)), HEADER_ROW)
                    If colTarget > 0 Then

                        v = CleanText(CStr(wsSrc.Cells(foundRow, colTarget).Value))

                        If replaceHyphenMode = "chiban" Then
                            v = NormalizeChiban(v)
                        End If

                        If v <> "" Then
                            If mergeRightCol <> "" Then
                                EnsureWritableRow_DE wsDst, outCol, mergeRightCol, outRow, outBaseEndRow
                            Else
                                EnsureWritableRow wsDst, outCol, outRow, outBaseEndRow
                            End If

                            WriteToMergedTopLeft wsDst, outCol, outRow, v
                            RecordBlockRow key, wsDst, outRow
                            outRow = outRow + GetMergedHeight(wsDst, outCol, outRow)
                        End If
                    End If

                Next i
            End If
        End If
    Next addr

CleanExit:
    Application.ScreenUpdating = prevScreenUpdating
    Application.EnableEvents = prevEnableEvents
    Application.Calculation = prevCalc
    Exit Sub

CleanFail:
    Application.ScreenUpdating = prevScreenUpdating
    Application.EnableEvents = prevEnableEvents
    Application.Calculation = prevCalc
    MsgBox "転記中にエラー: " & Err.Description, vbExclamation
End Sub

'======================================================================
' 【追加要件①】共通：接尾辞なしの単一見出しを縦詰め転記
' （例：契約年月日、申出日、竣工年月日など、①～⑩の接尾辞を持たない項目）
' 【修正②③】trigAddr 引数を追加し、指定された1つのトリガーのみを使用
'======================================================================
Private Sub Transfer_SingleHeader_Vertical( _
    ByVal outCol As String, _
    ByVal outStartRow As Long, _
    ByVal outBaseEndRow As Long, _
    ByVal headerName As String, _
    ByVal trigAddr As String)

    Dim wbSrc As Workbook, wsSrc As Worksheet
    Dim wsDst As Worksheet
    Dim prevScreenUpdating As Boolean, prevEnableEvents As Boolean, prevCalc As XlCalculation
    Dim colTransfer As Long, lastSrcRow As Long, foundRow As Long
    Dim key As String
    Dim outRow As Long
    Dim colTarget As Long
    Dim v As String

    prevScreenUpdating = Application.ScreenUpdating
    prevEnableEvents = Application.EnableEvents
    prevCalc = Application.Calculation

    On Error GoTo CleanFail
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    Set wsDst = ThisWorkbook.Worksheets(DST_SHEET)

    ' 【修正①】クリア：実際の最終行までクリア（残骸防止）
    Dim lastRowCol As Long, clearEnd As Long
    lastRowCol = wsDst.Cells(wsDst.rows.count, wsDst.Columns(outCol).Column).End(xlUp).Row
    clearEnd = WorksheetFunction.Max(outBaseEndRow, lastRowCol)
    ClearOutputMerged wsDst, outCol, outStartRow, clearEnd
    outRow = outStartRow

    Set wbSrc = GetOrOpenSourceWorkbook(SRC_FILE_EXACT, SRC_FILE_PATTERN)
    Set wsSrc = wbSrc.Worksheets(SRC_SHEET)

    colTransfer = FindHeaderCol(wsSrc, "転記", HEADER_ROW)
    If colTransfer = 0 Then Err.Raise vbObjectError + 100, , "参照元「表」シートに見出し「転記」が見つかりません。"

    lastSrcRow = wsSrc.Cells(wsSrc.rows.count, colTransfer).End(xlUp).Row
    If lastSrcRow < DATA_START_ROW Then GoTo CleanExit

    ' 見出し列を検索（接尾辞なし）
    colTarget = FindHeaderCol(wsSrc, headerName, HEADER_ROW)
    If colTarget = 0 Then GoTo CleanExit  ' 見出しが見つからない場合はスキップ（エラーにしない）

    ' 【修正②】指定された1つのトリガーのみを使用（TRIG_ADDRS を Split して回さない）
    key = NormalizeKey(wsDst.Range(trigAddr).Value)
    If key <> "" Then  ' 空白の場合は何も転記しない

        foundRow = FindRowByKey(wsSrc, colTransfer, key, DATA_START_ROW, lastSrcRow)
        If foundRow > 0 Then  ' foundRow が見つからない場合もスキップ（エラーにしない）

            v = CleanText(CStr(wsSrc.Cells(foundRow, colTarget).Value))

            If v <> "" Then
                ' 行挿入対応
                EnsureWritableRow wsDst, outCol, outRow, outBaseEndRow

                ' 結合セル対応：左上セルに書き込み
                WriteToMergedTopLeft wsDst, outCol, outRow, v
                outRow = outRow + GetMergedHeight(wsDst, outCol, outRow)
            End If
        End If
    End If

CleanExit:
    Application.ScreenUpdating = prevScreenUpdating
    Application.EnableEvents = prevEnableEvents
    Application.Calculation = prevCalc
    Exit Sub

CleanFail:
    Application.ScreenUpdating = prevScreenUpdating
    Application.EnableEvents = prevEnableEvents
    Application.Calculation = prevCalc
    MsgBox "転記中にエラー: " & Err.Description, vbExclamation
End Sub

'==================== キーの正規化（Trim/CStr で型ゆれを吸収） ====================
Private Function NormalizeKey(ByVal v As Variant) As String
    NormalizeKey = Trim$(CStr(v))
End Function

'==================== 見出し名の正規化（改行・空白揺れ対策） ====================
Private Function NormalizeHeader(ByVal v As Variant) As String
    Dim t As String
    t = CStr(v)
    t = Replace(t, vbCrLf, "")
    t = Replace(t, vbCr, "")
    t = Replace(t, vbLf, "")
    t = Replace(t, vbTab, "")
    t = Replace(t, " ", "")
    t = Replace(t, ChrW(12288), "")
    NormalizeHeader = Trim$(t)
End Function

'==================== 数字→全角数字 ====================
Private Function ToFullWidthNumber(ByVal n As Long) As String
    Dim s As String
    Dim i As Long
    Dim ch As String
    Dim codePoint As Long

    s = CStr(n)
    ToFullWidthNumber = ""
    For i = 1 To Len(s)
        ch = Mid$(s, i, 1)
        If ch >= "0" And ch <= "9" Then
            codePoint = AscW(ch) + &HFF10 - AscW("0")
            ToFullWidthNumber = ToFullWidthNumber & ChrW(codePoint)
        Else
            ToFullWidthNumber = ToFullWidthNumber & ch
        End If
    Next i
End Function

'==================== 【修正②】トリガーセルから最初の有効なキーを取得 ====================
' S1 → T1 → U1 の順で、最初に値が入っているトリガーセルの転記コードを取得
Private Function GetFirstTriggerKey(ByVal wsDst As Worksheet) As String
    Dim trigCells As Variant, addr As Variant
    Dim key As String
    
    trigCells = Split(TRIG_ADDRS, ",")
    For Each addr In trigCells
        key = NormalizeKey(wsDst.Range(CStr(addr)).Value)
        If key <> "" Then
            GetFirstTriggerKey = key
            Exit Function
        End If
    Next addr
    
    GetFirstTriggerKey = ""
End Function

'==================== 【修正④】地番表記の正規化：〇-□ → 〇番□ / 無ければ末尾に番 ====================
Private Function NormalizeChiban(ByVal s As String) As String
    Dim t As String
    t = Trim$(s)
    If t = "" Then
        NormalizeChiban = ""
        Exit Function
    End If

    ' ハイフン（-、－、‐、―）はすべて「番」に置換
    If (InStr(t, "-") > 0) Or (InStr(t, "－") > 0) Or (InStr(t, "‐") > 0) Or (InStr(t, "―") > 0) Then
        t = Replace(t, "－", "番")
        t = Replace(t, "-", "番")
        t = Replace(t, "‐", "番")
        t = Replace(t, "―", "番")
    Else
        ' ハイフンが存在しない場合は、末尾に「番」を付与
        If Right$(t, 1) <> "番" Then t = t & "番"
    End If

    NormalizeChiban = Trim$(t)
End Function

'==================== 参照元ブック取得（同一フォルダから自動検出） ====================
Private Function GetOrOpenSourceWorkbook(ByVal exactName As String, ByVal pattern As String) As Workbook
    Dim folderPath As String
    Dim fullExact As String, fullFound As String
    Dim fileFound As String
    Dim wb As Workbook

    folderPath = ThisWorkbook.path
    If folderPath = "" Then
        Err.Raise vbObjectError + 210, , "このブックが未保存のため、同一フォルダを特定できません。先に保存してください。"
    End If

    For Each wb In Application.Workbooks
        If LCase$(wb.name) = LCase$(exactName) Then
            Set GetOrOpenSourceWorkbook = wb
            Exit Function
        End If
    Next wb

    fullExact = folderPath & "\" & exactName
    If Dir(fullExact) <> "" Then
        Set GetOrOpenSourceWorkbook = Application.Workbooks.Open(fileName:=fullExact, ReadOnly:=True)
        Exit Function
    End If

    fileFound = Dir(folderPath & "\" & pattern)
    If fileFound = "" Then
        Err.Raise vbObjectError + 200, , _
            "参照元ブックが見つかりません。" & vbCrLf & _
            "検索フォルダ: " & folderPath & vbCrLf & _
            "探した名前: " & exactName & vbCrLf & _
            "探したパターン: " & pattern
    End If

    For Each wb In Application.Workbooks
        If LCase$(wb.name) = LCase$(fileFound) Then
            Set GetOrOpenSourceWorkbook = wb
            Exit Function
        End If
    Next wb

    fullFound = folderPath & "\" & fileFound
    Set GetOrOpenSourceWorkbook = Application.Workbooks.Open(fileName:=fullFound, ReadOnly:=True)
End Function

'==================== 見出し名→列番号 ====================
Private Function FindHeaderCol(ByVal ws As Worksheet, ByVal headerName As String, ByVal headerRow As Long) As Long
    Dim lastCol As Long, c As Long
    lastCol = ws.Cells(headerRow, ws.Columns.count).End(xlToLeft).Column
    For c = 1 To lastCol
        If NormalizeHeader(ws.Cells(headerRow, c).Value) = NormalizeHeader(headerName) Then
            FindHeaderCol = c
            Exit Function
        End If
    Next c
    FindHeaderCol = 0
End Function

'==================== キー一致行を検索（完全一致） ====================
Private Function FindRowByKey(ByVal ws As Worksheet, ByVal keyCol As Long, ByVal key As String, ByVal startRow As Long, ByVal lastRow As Long) As Long
    Dim r As Long
    Dim normKey As String
    Dim cellKey As String

    normKey = NormalizeKey(key)
    If normKey = "" Then
        FindRowByKey = 0
        Exit Function
    End If

    For r = startRow To lastRow
        cellKey = NormalizeKey(ws.Cells(r, keyCol).Value)
        If cellKey = normKey Then
            FindRowByKey = r
            Exit Function
        End If
    Next r

    FindRowByKey = 0
End Function

'==================== ブロック境界記録：初期化 ====================
Private Sub ResetBlockTracking()
    Set BlockRowsByTrigger = New Collection
End Sub

'==================== ブロック境界記録：行追加 ====================
Private Sub RecordBlockRow(ByVal key As String, ByVal ws As Worksheet, ByVal rowNum As Long)
    Dim rows As Collection
    Dim lastRange As Range

    If key = "" Then Exit Sub
    If BlockRowsByTrigger Is Nothing Then Set BlockRowsByTrigger = New Collection

    Set rows = GetBlockRowsByKey(key)
    If rows.count > 0 Then
        Set lastRange = rows(rows.count)
        If lastRange.Row = rowNum Then Exit Sub
    End If

    rows.Add ws.rows(rowNum)
End Sub

'==================== ブロック境界記録：キー別行コレクション取得 ====================
Private Function GetBlockRowsByKey(ByVal key As String) As Collection
    Dim rows As Collection

    On Error Resume Next
    Set rows = BlockRowsByTrigger(key)
    On Error GoTo 0

    If rows Is Nothing Then
        Set rows = New Collection
        BlockRowsByTrigger.Add rows, key
    End If

    Set GetBlockRowsByKey = rows
End Function

'==================== ブロック境界記録：範囲取得 ====================
Private Function GetBlockRangeByKey(ByVal ws As Worksheet, ByVal key As String, ByRef blockStart As Long, ByRef blockEnd As Long) As Boolean
    Dim rows As Collection
    Dim firstRange As Range
    Dim lastRange As Range
    Dim lastRow As Long

    If BlockRowsByTrigger Is Nothing Then Exit Function

    On Error Resume Next
    Set rows = BlockRowsByTrigger(key)
    On Error GoTo 0

    If rows Is Nothing Then Exit Function
    If rows.count = 0 Then Exit Function

    Set firstRange = rows(1)
    Set lastRange = rows(rows.count)
    blockStart = firstRange.Row
    lastRow = lastRange.Row
    blockEnd = lastRow + GetMergedHeightMulti(ws, lastRow, Array("B", "C", "D", "F", "G", "I", "K", "M"))
    GetBlockRangeByKey = True
End Function

'==================== 結合対応：出力欄クリア（MergeArea単位） ====================
Private Sub ClearOutputMerged(ByVal ws As Worksheet, ByVal colLetter As String, ByVal startRow As Long, ByVal endRow As Long)
    Dim r As Long
    Dim c As Range, ma As Range
    Dim colNum As Long
    colNum = ws.Columns(colLetter).Column

    r = startRow
    Do While r <= endRow
        Set c = ws.Cells(r, colNum)
        If c.MergeCells Then
            Set ma = c.MergeArea
            ma.ClearContents
            r = ma.Row + ma.rows.count
        Else
            c.ClearContents
            r = r + 1
        End If
    Loop
End Sub

'==================== 出力先：基本枠超え/結合なら行挿入して回避（単列） ====================
' 【修正①】結合セルであることだけでは行挿入しない。既に値が入っている場合のみ行挿入する。
Private Sub EnsureWritableRow(ByVal ws As Worksheet, ByVal colLetter As String, ByVal rowNum As Long, ByVal baseEndRow As Long)
    Dim colNum As Long
    Dim c As Range
    Dim hasValue As Boolean

    colNum = ws.Columns(colLetter).Column
    Set c = ws.Cells(rowNum, colNum)

    ' 基本枠を超えている場合は行挿入
    If rowNum > baseEndRow Then
        ws.rows(rowNum).Insert Shift:=xlShiftDown, CopyOrigin:=xlFormatFromLeftOrAbove
        Exit Sub
    End If

    ' 既に値が入っているかチェック（結合セルの場合は左上セルを確認）
    If c.MergeCells Then
        hasValue = (Trim$(CStr(c.MergeArea.Cells(1, 1).Value)) <> "")
    Else
        hasValue = (Trim$(CStr(c.Value)) <> "")
    End If

    ' 既に値が入っている場合のみ行挿入（結合セルであることだけでは行挿入しない）
    If hasValue Then
        ws.rows(rowNum).Insert Shift:=xlShiftDown, CopyOrigin:=xlFormatFromLeftOrAbove
    End If
End Sub

'==================== 出力先：基本枠超え/結合なら行挿入して回避（複数列） ====================
' 【修正①】結合セルであることだけでは行挿入しない。既に値が入っている場合のみ行挿入する。
Private Sub EnsureWritableRowMulti(ByVal ws As Worksheet, ByVal rowNum As Long, ByVal baseEndRow As Long, ByVal colLetters As Variant)
    Dim i As Long, colNum As Long
    Dim c As Range
    Dim hasValue As Boolean

    ' 基本枠を超えている場合は行挿入
    If rowNum > baseEndRow Then
        ws.rows(rowNum).Insert Shift:=xlShiftDown, CopyOrigin:=xlFormatFromLeftOrAbove
        Exit Sub
    End If

    ' いずれかの列に既に値が入っているかチェック
    For i = LBound(colLetters) To UBound(colLetters)
        colNum = ws.Columns(CStr(colLetters(i))).Column
        Set c = ws.Cells(rowNum, colNum)
        
        ' 結合セルの場合は左上セルを確認
        If c.MergeCells Then
            hasValue = (Trim$(CStr(c.MergeArea.Cells(1, 1).Value)) <> "")
        Else
            hasValue = (Trim$(CStr(c.Value)) <> "")
        End If
        
        ' 既に値が入っている場合のみ行挿入（結合セルであることだけでは行挿入しない）
        If hasValue Then
            ws.rows(rowNum).Insert Shift:=xlShiftDown, CopyOrigin:=xlFormatFromLeftOrAbove
            Exit Sub
        End If
    Next i
End Sub

'==================== 出力先：D:E横結合維持 ====================
' 【修正①】結合セルであることだけでは行挿入しない。既に値が入っている場合のみ行挿入する。
Private Sub EnsureWritableRow_DE(ByVal ws As Worksheet, ByVal leftCol As String, ByVal rightCol As String, ByVal rowNum As Long, ByVal baseEndRow As Long)
    Dim c As Range
    Dim hasValue As Boolean
    
    Set c = ws.Cells(rowNum, ws.Columns(leftCol).Column)

    ' 基本枠を超えている場合は行挿入
    If rowNum > baseEndRow Then
        ws.rows(rowNum).Insert Shift:=xlShiftDown, CopyOrigin:=xlFormatFromLeftOrAbove
        MergeDEIfNeeded ws, leftCol, rightCol, rowNum
        Exit Sub
    End If

    ' 既に値が入っているかチェック（結合セルの場合は左上セルを確認）
    If c.MergeCells Then
        hasValue = (Trim$(CStr(c.MergeArea.Cells(1, 1).Value)) <> "")
    Else
        hasValue = (Trim$(CStr(c.Value)) <> "")
    End If

    ' 既に値が入っている場合、または D:E の横結合ではない結合セルの場合は行挿入
    If hasValue Then
        ws.rows(rowNum).Insert Shift:=xlShiftDown, CopyOrigin:=xlFormatFromLeftOrAbove
        MergeDEIfNeeded ws, leftCol, rightCol, rowNum
    ElseIf c.MergeCells Then
        ' D:E の横結合ではない結合セルの場合も行挿入（既存仕様維持）
        If c.MergeArea.Columns.count <> 2 Or c.MergeArea.Column <> ws.Columns(leftCol).Column Then
            ws.rows(rowNum).Insert Shift:=xlShiftDown, CopyOrigin:=xlFormatFromLeftOrAbove
            MergeDEIfNeeded ws, leftCol, rightCol, rowNum
        Else
            ' D:E の横結合の場合は結合を維持
            MergeDEIfNeeded ws, leftCol, rightCol, rowNum
        End If
    Else
        ' 結合セルでない場合は D:E を結合
        MergeDEIfNeeded ws, leftCol, rightCol, rowNum
    End If
End Sub

Private Sub MergeDEIfNeeded(ByVal ws As Worksheet, ByVal leftCol As String, ByVal rightCol As String, ByVal rowNum As Long)
    Dim rngDE As Range
    Set rngDE = ws.Range(leftCol & rowNum & ":" & rightCol & rowNum)

    On Error Resume Next
    If rngDE.MergeCells = False Then
        rngDE.Merge
    Else
        If rngDE.MergeArea.Address <> rngDE.Address Then
            rngDE.MergeArea.UnMerge
            rngDE.Merge
        End If
    End If
    On Error GoTo 0
End Sub

'==================== 結合対応：結合の高さ（単列） ====================
Private Function GetMergedHeight(ByVal ws As Worksheet, ByVal colLetter As String, ByVal rowNum As Long) As Long
    Dim colNum As Long
    Dim c As Range
    colNum = ws.Columns(colLetter).Column
    Set c = ws.Cells(rowNum, colNum)

    Else
        GetMergedHeight = 1
    End If

'==================== 結合対応：結合の高さ（複数列の最大） ====================
Private Function GetMergedHeightMulti(ByVal ws As Worksheet, ByVal rowNum As Long, ByVal colLetters As Variant) As Long
    Dim i As Long, h As Long, maxH As Long
    Dim colNum As Long
    Dim c As Range

    maxH = 1
    For i = LBound(colLetters) To UBound(colLetters)
        colNum = ws.Columns(CStr(colLetters(i))).Column
        Set c = ws.Cells(rowNum, colNum)
        If c.MergeCells Then h = c.MergeArea.rows.count Else h = 1
        If h > maxH Then maxH = h
    Next i

    GetMergedHeightMulti = maxH
End Function
